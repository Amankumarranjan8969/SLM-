def factorial(n: int) -> int:
    """Return n! computed iteratively, avoiding recursion depth limits."""
    if n < 0:
        raise ValueError("factorial is undefined for negative integers")
    result = 1
    for i in range(2, n + 1):
        result *= i
    return result


def is_prime(n: int) -> bool:
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    i = 3
    while i * i <= n:
        if n % i == 0:
            return False
        i += 2
    return True


class BinarySearchTree:
    """A minimal, unbalanced binary search tree for teaching purposes."""

    def __init__(self):
        self.root = None

    def insert(self, value):
        if self.root is None:
            self.root = _Node(value)
            return
        node = self.root
        while True:
            if value < node.value:
                if node.left is None:
                    node.left = _Node(value)
                    return
                node = node.left
            else:
                if node.right is None:
                    node.right = _Node(value)
                    return
                node = node.right


class _Node:
    __slots__ = ("value", "left", "right")

    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None


// A small C-style function for computing the greatest common divisor.
int gcd(int a, int b) {
    while (b != 0) {
        int t = b;
        b = a % b;
        a = t;
    }
    return a;
}

public class QuickSort {
    public static void sort(int[] arr, int low, int high) {
        if (low < high) {
            int pivotIndex = partition(arr, low, high);
            sort(arr, low, pivotIndex - 1);
            sort(arr, pivotIndex + 1, high);
        }
    }

    private static int partition(int[] arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;
        for (int j = low; j < high; j++) {
            if (arr[j] <= pivot) {
                i++;
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;
        return i + 1;
    }
}

Machine learning notes:
A neural network layer applies a linear transformation followed by a
nonlinear activation function such as ReLU, GELU, or SiLU. Gradient
descent updates each parameter in the direction that most reduces the
loss function, scaled by a learning rate. Overfitting occurs when a
model memorizes its training data instead of learning patterns that
generalize to unseen examples; regularization, dropout, and early
stopping are common ways to reduce it. Attention mechanisms let a
model weigh the relevance of every token in a sequence when producing
a representation for any single token, which is why transformers can
model long-range dependencies more directly than recurrent networks.

Data structures and algorithms glossary:
- Stack: last-in, first-out access; push and pop are O(1).
- Queue: first-in, first-out access; enqueue and dequeue are O(1)
  with a proper circular buffer or linked list implementation.
- Hash table: average O(1) lookup, insertion, and deletion, backed by
  an array of buckets and a hash function that distributes keys
  uniformly.
- Binary search: O(log n) lookup in a sorted array by repeatedly
  halving the search interval.
- Depth-first search and breadth-first search both visit every
  reachable node in a graph, but in different orders and using
  different auxiliary data structures (a stack versus a queue).
